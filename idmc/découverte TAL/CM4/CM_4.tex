\begin{frame}{Limites des données textuelles}
    \vfill
    \begin{block}{Limites}
        \begin{itemize}
            \item Pas d'analyse sémantique entre les mots
            \item Peu d'analyse relationnelle entre les mots
            \item Les modèles statistiques ont besoin de données numériques
        \end{itemize}
    \end{block}
    \vfill
    \begin{center}
        \Large
        Solution~: représenter différemment les mots ! 
    \end{center}
\end{frame}

\begin{frame}{Principes de la vectorisation}
    \begin{block}{Objectifs}
        \begin{itemize}
            \item Transformer chaque mot en une suite de chiffres $\Rightarrow$ vecteur
            \item Capturer des informations contextuelles pour chaque mot
            \item Capturer des relations entre les mots
        \end{itemize}
    \end{block}
    \vfill
    \begin{block}{Deux intuitions}
        \begin{itemize}
            \item Un mot = composé de plusieurs facteurs implicites
            \begin{itemize}
                \item chat = 10 (animal) + 5 (doux) - 10 (loyal)
            \end{itemize}
            \vspace{5pt}
            \item Les vecteurs de deux mots proches devraient être proches
            \begin{itemize}
                \item "\textit{You shall know a word by the company it keeps}" (Firth, 1957)
            \end{itemize}
        \end{itemize}
    \end{block}
    % \vfill
    % \begin{center}
    %     \Large
    %     Comment vectoriser du texte ? \\
    %     Que contiennent les vecteurs créés ?
    % \end{center}
\end{frame}

\begin{frame}{Pour résumer~: matrices de co-occurrences et sacs de mots}
    \begin{block}{Avantages}
        \begin{itemize}
            \item Très simples à créer (on compte des occurrences)
            \item Compréhensibles par des humains
        \end{itemize}
    \end{block}
    \vfill
    \begin{block}{Inconvénients}
        \begin{itemize}
            \item Taille des vecteurs = taille du vocabulaire
            \item Quid des mots polysémiques ?
            \item Quid des variantes orthographiques ?
            \item Quid des expressions multi-mots ?
            \item Quid des néologismes (mots non-vus) ?
        \end{itemize}
    \end{block}
\end{frame}

\section{Vectorisation~: pour aller plus loin}

\begin{exoframe}
    \begin{center}
        \Large
        La vectorisation ne s'arrête pas au \textbf{comptage} de \textbf{mots} ! \\
        \vspace{20pt}
        Nous ne sommes pas limités aux \textbf{sacs de mots}. \\
        On peut aller plus loin qu'un simple \textbf{comptage}. 
    \end{center}
\end{exoframe}

\begin{frame}{Représentation par sacs de N-grammes}
    \pause
    Rien ne sert de courir, il faut partir à point
    \vfill
    \begin{block}{N-grammes de caractères}
        \begin{itemize}
            \item Bi-grammes~: \wordbox{ri} \wordbox{ie} \wordbox{en} \wordbox{nn} \wordbox{ne}...
            \item Tri-grammes~: \wordbox{rie} \wordbox{ien} \wordbox{enn} \wordbox{nne} \wordbox{nes}...
        \end{itemize}
    \end{block}
    \vfill
    \begin{block}{N-grammes de mots/tokens}
        \begin{itemize}
            \item Bi-grammes~: \wordbox{rien ne} \wordbox{ne sert} \wordbox{sert de} \wordbox{de courir} \wordbox{courir il}...
        \end{itemize}
    \end{block}
    \vfill
    \begin{center}
        \Large
        C'est dur ? Utilisez l'astuce de la fenêtre !
    \end{center}
\end{frame}

\begin{frame}{Pondération TF-IDF}
    \begin{block}{Intuitions}
        \begin{itemize}
            \item Compter des mots, c'est quand même un peu limité
            \item On peut remplacer ce comptage par un calcul
        \end{itemize}
    \end{block}
    \vfill
    \begin{block}{Pondération TF-IDF}
        \begin{itemize}
            \item Mesure statistique (bouh des maths)
            \item Évalue l'importance d'un terme dans un corpus
            \item Combine deux mesures~: le TF et l'IDF
        \end{itemize}
    \end{block}
    \vfill
    \begin{center}
        \Large
        Le TF-IDF permet de pondérer chaque mot. \\
        Un mot très souvent utilisé aura un TF-IDF important.
    \end{center}
\end{frame}

\begin{frame}{Comment la calculer ?}
    \begin{minipage}{0.3\columnwidth}
        \begin{center}
            TF \\ (term frequency)
            \begin{equation*}
                \frac{N~terme~dans~document}{N~mots~dans~document}
            \end{equation*}
        \end{center}
    \end{minipage}
    \hspace{15pt}
    \begin{minipage}{0.35\columnwidth}
        \begin{center}
            IDF \\ (inverted doc. frequency)
            \begin{equation*}
                log(\frac{N~documents}{N~documents~avec~terme})
            \end{equation*}
        \end{center}
    \end{minipage}
    \hspace{45pt}
    \begin{minipage}{0.1\columnwidth}
        \begin{center}
            TF-IDF \\ \textcolor{white}{TF-IDF}
            \begin{equation*}
                TF~\cdot~IDF
            \end{equation*}
        \end{center}
    \end{minipage}
    \vfill
    \begin{block}{En français}
        \begin{itemize}
            \item TF~: fréquence du terme dans un document
            \item IDF~: inverse de la répartition du terme dans un corpus
            \item TF-IDF~: le produit des deux
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Pour résumer~: vectorisation}
    \begin{block}{Avantages}
        \begin{itemize}
            \item Capture d'informations contextuelles pour chaque mot
            \item Capture des relations entre les mots
            \item Transforme des données textuelles en données numériques
        \end{itemize}
    \end{block}
    \vfill
    \begin{block}{Inconvénients}
        \begin{itemize}
            \item Taille d'un vecteur = taille du vocabulaire
            \item Il faut trouver des moyens de réduire la taille des vecteurs
        \end{itemize}
    \end{block}
\end{frame}

\begin{exoframe}{Remarque importante}
    \begin{center}
        \Large
        Les vecteurs ne sont que des suites de chiffres ! \\
        Ils stockent les informations contextuelles de chaque mot.
    \end{center}
    \vfill
    \begin{block}{En conséquence~:}
        \begin{itemize} \color{white}
            \item Vecteur d'un mot $\ne$ définition d'un mot
            \item Les représentations sont biaisées par le contenu des corpus
        \end{itemize}
    \end{block}
\end{exoframe}
