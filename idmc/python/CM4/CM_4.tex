\begin{frame}{Respect the Rules}
    \begin{itemize}
        \item Here is the intented name of a notebook: TD1\_LASTNAME\_LASTNAME.ipynb
        \item A TD without your name $\Rightarrow$ points denied
        \item A TD without the name of your binom $\Rightarrow$ points denied
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Do What I Ask for}
    \begin{itemize}
        \item "Write a function to..." $\Rightarrow$ function expected
    \end{itemize}
    \begin{pythoncode}
    a = 1
    b = 2

    my_add_function = ( a + b)
    \end{pythoncode}
\end{frame}

\begin{frame}[fragile]{Identation Error}
    \begin{itemize}
        \item Stick to one identation scheme (2 spaces or 4 spaces for instance, not both)
    \end{itemize}
    \begin{pythoncode}
    def my_add_function(a,b):
        return a + b

    def my_add_function(a,b):
      return a + b
    \end{pythoncode}
\end{frame}

\begin{frame}[fragile]{Identation Error II, the Return}
    \begin{itemize}
        \item "return" should not be in a loop, unless you add a condition 
    \end{itemize}
    \begin{pythoncode}
    def find_mystery_number():
        guess = 1
        while hot_or_cold(guess) != "bingo bango!":
            guess += 1 
            return guess
    \end{pythoncode}
\end{frame}

\begin{frame}[fragile]{Variables Names Matters}
    \begin{itemize}
        \item Don't use variable names which are used in Python builds-in
    \end{itemize}
    \begin{pythoncode}
    list = []
    list.append(i)
    len(list)
    list[i]
    \end{pythoncode}
\end{frame}

\begin{frame}[fragile]{Function Calling}
    \begin{itemize}
        \item Be careful with function calling, names must be respected
    \end{itemize}
    \begin{pythoncode}
    def convert_fahr_to_cels(def_fahr):
        # do stuff...

    fahrenheit_to_celsius(fahrenheit)
    \end{pythoncode}
\end{frame}

\begin{frame}[fragile]{"Inspiration" VS plagiarism}
    \begin{itemize}
        \item DO NOT COPY/PASTE SOMEONE ELSE CODE !!!
    \end{itemize}
    \begin{pythoncode}
    # Influenced by "some url"

    def collatz_number(n):
        # do stuff...
    \end{pythoncode}
\end{frame}

\begin{frame}[fragile]{Infinite Loop} %error if number < 10
    \begin{pythoncode}
    def find_mystery_number(): 
        guess = 10
        while True:
            result = hot_or_cold(guess)
            if result == "bingo bango!":
                return guess
            if result == "you're hot!":
                break
            guess += 9
        guess -= 4
        while True:
            result = hot_or_cold(guess)
            if result == "bingo bango!":
                return guess    
            guess += 1
    \end{pythoncode}
