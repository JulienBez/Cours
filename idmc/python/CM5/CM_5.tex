\begin{frame}{What is a File Format ?}
    \begin{block}{Wikipedia Definition\footnote{\url{https://en.wikipedia.org/wiki/File_format}}}
        \begin{tcolorbox}
            A file format is the way that information is encoded for storage in a computer file. It may describe the encoding at various levels of abstraction including low-level bit and byte layout as well high-level organization such as markup and tabular structure. A file format may be standardized (which can be proprietary or open) or it can be an ad hoc convention.
        \end{tcolorbox}
    \end{block}
    \begin{center}
        \Large
        File format define the the structure of a file. \\
        It can be standardized or ad hoc.
    \end{center}
    %human-readable vs machine readable format
    %compromises = can be read by humans but has a strict structure to facilitate navigation with a machine
    % example : markup languages => set of rules to combine with desired content to create a format both readable by a human and easily manipulable by a machine
    %add image of a codebarre + image of a markup language (HTML ?) + image of human readable format 
\end{frame}

\begin{frame}{What is a Filename Extension ?}
    \begin{block}{Wikipedia Definition\footnote{\url{https://en.wikipedia.org/wiki/Filename_extension}}}
       \begin{tcolorbox}
        A filename extension, also known as file name extension or file extension, is a suffix to the name of a computer file that indicates a characteristic of the file contents or its intended use.
       \end{tcolorbox} 
    \end{block}
    \begin{center}
        \Large
        Filename extension indicates the format of a file to your computer. \\
        Some examples: .txt, .mp3, .parquet, .json, etc
    \end{center}
\end{frame}

\begin{frame}{What is a Character Encoding ?}
    \begin{block}{Wikipedia Definition\footnote{\url{https://en.wikipedia.org/wiki/Character_encoding}}}
        \begin{tcolorbox}
            Character encoding is a convention of using a numeric value to represent each character of a writing script. The most popular character encoding on the World Wide Web is UTF-8, which is used in 99.0\% of surveyed web sites, as of September 2026.
        \end{tcolorbox}
    \end{block}
    \begin{center}
        \Large
        Character encoding define how each character is \\ encoded in bits and decoded from bits.
    \end{center}
\end{frame}

\section{Handling a File in Python}

\begin{frame}[fragile]{File Opening and Writing (1/2)}
    \begin{pythoncode}
    with open("filepath.txt","r") as f:
        content = f.read() # load filepath.txt content

    with open("filepath.txt","w") as f:
        f.write(content) # save content to filepath.txt
    \end{pythoncode}
    \vfill
    \begin{block}{What would happen if...}
        \begin{itemize}
            \item I change the filename extension ?
            \item I change the file format ?
            \item I change the encoding of the file ?
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{File Opening and Writing (2/2)}
    \begin{block}{Changing filename extension}
        \begin{itemize}
            \item Will not change anything
            \item \wordbox{open()} always read a file as a string (or a list of strings)
        \end{itemize}
    \end{block}
    \begin{block}{Changing file format}
        \begin{itemize}
            \item Will not change anything
            \item Problem: everything will be in a string (data structure is "lost")
        \end{itemize}
    \end{block}
    \begin{block}{Changing the encoding}
        \begin{itemize}
            \item May result in incorrect reading of the file
            \item Solution: use \wordbox{encoding='your\_file\_encoding'} in \wordbox{open()}
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{First Warning: Encoding}
    \begin{block}{Encoding mistakes are pretty common}
        \begin{itemize}
            \item Linux and Mac: UTF-8
            \item Windows: Unicode or ANSI
        \end{itemize}
    \end{block}
    \begin{block}{How to find and fix them}
        \begin{itemize}
            \item Check file encoding before opening it
            \item Open your file and print some content from it
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Second Warning: Pathing}
    \begin{block}{Two kinds of path}
        \begin{itemize}
            \item Relative path
            \begin{itemize}
                \item my\_data\_folder/file.txt
            \end{itemize}
            \item Absolute path
            \begin{itemize}
                \item /home/documents/my\_data\_folder/file.txt
            \end{itemize}
        \end{itemize}
    \end{block}
    \begin{block}{Path syntax may vary}
        \begin{itemize}
            \item Linux: uses slashes (/) to navigate through folders
            \item Windows: uses backslashes (\bs) to navigate through folders
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{String Parsing}
    \begin{center}
        \Large
        You can use string methods to extract structured data from a string.
    \end{center}
    \begin{block}{How to do it ?}
        \begin{itemize}
            \item By understanding the file structure

    \begin{block}{Disadvantages}

    content = """"
    name,username,id,description
    toto,toto78,123,his name is toto

    """ # this is a csv file structure

    structured = content.split("\n") # separate each line
    structured = [i.split(",") for i in structured]

    print(structured)

\begin{frame}[fragile]{Folder Management}
    \begin{pythoncode}
    # Create a folder
    if not os.path.exists("my_path"):
        os.makedir("my_path")

    # Create multiple folders
    Create a folder
    if not os.path.exists("my_folder/my_path"):
        os.makedirs("my_folder/my_path")

    # Delete a file or a folder
    os.remove("path_to_file")
    os.rmdir("path_to_folder")
    \end{pythoncode}
\end{frame}

\begin{frame}[fragile]{Folder Inspection}
    \begin{pythoncode}
    # Get all paths of files in a folder with os
    import os
    path_list = os.listdir("path")

    # Get all paths of files in a folder with glob
    import glob
    path_list = [path for path in glob.glob("path/*.txt")]

    # Get all paths of files in folder and subfolders
    import os
    for path, subdirs, files in os.walk("path"):
        print(path,subdirs,files)
    \end{pythoncode}
\end{frame}

\begin{frame}[fragile]{Looping Through Files}
    \begin{pythoncode}
    import glob

    for path in glob.glob("my_path/*.txt"):
        print(path)
        # process the file
    \end{pythoncode}
    \vfill
    \begin{block}{General tips}
        \begin{itemize}
            \item Don't overwrite files - make copies !
            \item Always make sure the file is well-structured first
            \item Use \wordbox{try} and \wordbox{expect} to handle possible errors
        \end{itemize}
    \end{block}
\end{frame}
